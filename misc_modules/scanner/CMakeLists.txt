cmake_minimum_required(VERSION 3.13)
project(scanner)

file(GLOB SRC "src/*.cpp")

include(${SDRPP_MODULE_CMAKE})

target_include_directories(scanner PRIVATE "src/" "../../decoder_modules/radio/src")
# Speech recognition of transmissions, when whisper.cpp is installed (brew install whisper-cpp)
option(OPT_SCANNER_TRANSCRIPTION "Transcribe transmissions with whisper.cpp when it is found" ON)
if (OPT_SCANNER_TRANSCRIPTION)
    find_package(PkgConfig)
    pkg_check_modules(WHISPER whisper ggml)
    if (WHISPER_FOUND)
        target_compile_definitions(scanner PRIVATE SCANNER_TRANSCRIPTION)
        target_include_directories(scanner PRIVATE ${WHISPER_INCLUDE_DIRS})
        target_link_directories(scanner PRIVATE ${WHISPER_LIBRARY_DIRS})
        target_link_libraries(scanner PRIVATE ${WHISPER_LIBRARIES})
    endif ()
endif ()
